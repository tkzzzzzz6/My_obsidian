typedef struct
{
    ElemType data[MaxSize];     //顺序栈需要设置最大值

}SqStack;
//入栈
bool Push(SqStack *&s,ElemType e)
{
    if(s->top==MaxSize-1)       //判断栈满则溢出
        return false;
    s->top++;                   //栈顶指针＋1
    s->data[s->top]=e;          //元素放指针处
    return true;
}
//出栈
bool Pop(SqStack *&s,ElemType &e)
{
    if(s->top==-1)              //栈为空，即栈下溢出
        return false;
    e=s->data[s->top];             //取栈顶元素
    s->top--;                       //栈顶指针-1
    return true;
}
